    77→function formatBytes(bytes: number): string {
    78→  if (bytes === 0) return '0 B';
    79→  const k = 1024;
    80→  const sizes = ['B', 'KB', 'MB', 'GB'];
    81→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    82→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
    83→}

    85→function formatDate(dateStr: string): string {
    86→  return new Date(dateStr).toLocaleDateString('es-ES', {
    87→    day: 'numeric',
    88→    month: 'short',
    89→    year: 'numeric',
    90→    hour: '2-digit',
    91→    minute: '2-digit',
    92→  });
    93→}

    95→function getFileIcon(mimeType: string) {
    96→  if (mimeType.startsWith('image/')) return <FileImage className="h-8 w-8 text-emerald-500" />;
    97→  if (mimeType.startsWith('video/')) return <FileVideo className="h-8 w-8 text-violet-500" />;
    98→  if (mimeType.startsWith('audio/')) return <FileAudio className="h-8 w-8 text-amber-500" />;
    99→  if (mimeType.includes('zip') || mimeType.includes('rar') || mimeType.includes('tar') || mimeType.includes('gz'))
   100→    return <Archive className="h-8 w-8 text-orange-500" />;
   101→  if (mimeType.includes('pdf')) return <FileText className="h-8 w-8 text-red-500" />;
   102→  if (mimeType.includes('text') || mimeType.includes('json') || mimeType.includes('xml'))
   103→    return <FileText className="h-8 w-8 text-sky-500" />;
   104→  return <File className="h-8 w-8 text-muted-foreground" />;
   105→}

   107→function getFileExtension(name: string): string {
   108→  const parts = name.split('.');
   109→  return parts.length > 1 ? parts.pop()!.toUpperCase() : 'N/A';

   329→/* ──────────── Upload Dialog ──────────── */
   330→function UploadDialog({ onUploaded }: { onUploaded: () => void }) {
   331→  const [files, setFiles] = useState<File[]>([]);
   332→  const [uploading, setUploading] = useState(false);
   333→  const [progress, setProgress] = useState(0);
   334→  const [dragActive, setDragActive] = useState(false);
   335→  const fileInputRef = useRef<HTMLInputElement>(null);
   336→  const { toast } = useToast();
   337→
   338→  const handleFiles = (fileList: FileList | null) => {
   339→    if (!fileList) return;
   340→    setFiles((prev) => [...prev, ...Array.from(fileList)]);
   341→  };
   342→
   343→  const handleDrop = useCallback((e: React.DragEvent) => {
   344→    e.preventDefault();
   345→    setDragActive(false);
   346→    handleFiles(e.dataTransfer.files);
   347→  }, []);
   348→
   349→  const handleDragOver = (e: React.DragEvent) => {
   350→    e.preventDefault();
   351→    setDragActive(true);
   352→  };
   353→
   354→  const handleDragLeave = () => setDragActive(false);
   355→
   356→  const removeFile = (index: number) => {
   357→    setFiles((prev) => prev.filter((_, i) => i !== index));
   358→  };
   359→
   360→  const upload = async () => {
   361→    if (files.length === 0) return;
   362→    setUploading(true);
   363→    setProgress(0);
   364→
   365→    let success = 0;
   366→    let failed = 0;
   367→
   368→    for (let i = 0; i < files.length; i++) {
   369→      const file = files[i];
   370→      try {
   371→        // Step 1: Get presigned URL from our API (small request, no file data)
   372→        const presignRes = await fetch('/api/files/presign', {
   373→          method: 'POST',
   374→          headers: { 'Content-Type': 'application/json' },
   375→          body: JSON.stringify({
   376→            fileName: file.name,
   377→            fileSize: file.size,
   378→            mimeType: file.type || 'application/octet-stream',
   379→          }),
   380→        });
   381→
   382→        if (!presignRes.ok) {
   383→          failed++;
   384→          setProgress(Math.round(((i + 1) / files.length) * 100));
   385→          continue;
   386→        }
   387→
   388→        const { uploadUrl, r2Key } = await presignRes.json();
   389→
   390→        // Step 2: Upload file DIRECTLY to R2 (bypasses Vercel 4.5MB limit)
   391→        const uploadRes = await fetch(uploadUrl, {
   392→          method: 'PUT',
   393→          body: file,
   394→          headers: { 'Content-Type': file.type || 'application/octet-stream' },
   395→        });
   396→
   397→        if (!uploadRes.ok) {
   398→          failed++;
   399→          setProgress(Math.round(((i + 1) / files.length) * 100));
   400→          continue;
   401→        }
   402→
   403→        // Step 3: Confirm - save metadata to database
   404→        const confirmRes = await fetch('/api/files/confirm', {
   405→          method: 'POST',
   406→          headers: { 'Content-Type': 'application/json' },
   407→          body: JSON.stringify({
   408→            r2Key,
   409→            originalName: file.name,
   410→            mimeType: file.type || 'application/octet-stream',
   411→            size: file.size,
   412→          }),
   413→        });
   414→
   415→        if (confirmRes.ok) success++;
   416→        else failed++;
   417→      } catch {
   418→        failed++;
   419→      }
   420→      setProgress(Math.round(((i + 1) / files.length) * 100));
   421→    }
   422→
   423→    setUploading(false);
   424→    setFiles([]);
   425→
   426→    if (success > 0) {
   427→      toast({
   428→        title: `${success} archivo${success > 1 ? 's' : ''} subido${success > 1 ? 's' : ''}`,
   429→        description: failed > 0 ? `${failed} archivo${failed > 1 ? 's' : ''} falló.` : 'Todos los archivos se subieron correctamente.',
   430→        variant: failed > 0 ? 'default' : 'default',
   431→      });
   432→      onUploaded();
   433→    }
   434→
   435→    if (failed > 0 && success === 0) {
   436→      toast({
   437→        title: 'Error',
   438→        description: 'No se pudieron subir los archivos.',
   439→        variant: 'destructive',
   440→      });
   441→    }
   442→  };
   443→
   444→  return (
   445→    <Dialog>
   446→      <DialogTrigger asChild>
   447→        <Button className="bg-emerald-600 hover:bg-emerald-700 text-white font-medium h-11 gap-2">
   448→          <Upload className="h-4 w-4" />
   449→          Subir archivos
   450→        </Button>
   451→      </DialogTrigger>
   452→      <DialogContent className="sm:max-w-lg">
   453→        <DialogHeader>
   454→          <DialogTitle>Subir archivos</DialogTitle>
   455→          <DialogDescription>
   456→            Arrastra archivos aquí o haz clic para seleccionar. Máximo 5GB por archivo.
   457→          </DialogDescription>
   458→        </DialogHeader>
   459→
   460→        <div
   461→          className={`border-2 border-dashed rounded-xl p-8 text-center transition-colors cursor-pointer ${
   462→            dragActive
   463→              ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30'
   464→              : 'border-muted-foreground/25 hover:border-emerald-400 hover:bg-muted/50'
   465→          }`}
   466→          onDrop={handleDrop}
   467→          onDragOver={handleDragOver}
   468→          onDragLeave={handleDragLeave}
   469→          onClick={() => fileInputRef.current?.click()}
   470→        >
   471→          <CloudUpload className="h-10 w-10 mx-auto text-muted-foreground mb-3" />
   472→          <p className="text-sm font-medium">Arrastra archivos aquí</p>
   473→          <p className="text-xs text-muted-foreground mt-1">o haz clic para seleccionar</p>
   474→          <input
   475→            ref={fileInputRef}
   476→            type="file"
   477→            multiple
   478→            className="hidden"
   479→            onChange={(e) => handleFiles(e.target.files)}
   480→          />
   481→        </div>
   482→
   483→        {files.length > 0 && (
   484→          <div className="space-y-2 max-h-48 overflow-y-auto">
   485→            {files.map((f, i) => (
   486→              <div key={`${f.name}-${i}`} className="flex items-center gap-3 p-2 rounded-lg bg-muted/50">
   487→                {getFileIcon(f.type)}
   488→                <div className="flex-1 min-w-0">
   489→                  <p className="text-sm font-medium truncate">{f.name}</p>
   490→                  <p className="text-xs text-muted-foreground">{formatBytes(f.size)}</p>
   491→                </div>
   492→                <Button
   493→                  variant="ghost"
   494→                  size="icon"
   495→                  className="h-8 w-8 shrink-0"
   496→                  onClick={(e) => {
   497→                    e.stopPropagation();
   498→                    removeFile(i);
   499→                  }}
   500→                  disabled={uploading}
   501→                >
   502→                  <X className="h-4 w-4" />
   503→                </Button>
   504→              </div>
   505→            ))}
   506→          </div>
   507→        )}
   508→
   509→        {uploading && (
   510→          <div className="space-y-2">
   511→            <Progress value={progress} className="h-2" />
   512→            <p className="text-xs text-center text-muted-foreground">{progress}% completado</p>
   513→          </div>
   514→        )}
   515→
   516→        <div className="flex justify-end gap-2">
   517→          <Button
   518→            variant="outline"
   519→            onClick={() => setFiles([])}
   520→            disabled={uploading || files.length === 0}
   521→          >
   522→            Limpiar
   523→          </Button>
   524→          <Button
   525→            onClick={upload}
   526→            disabled={uploading || files.length === 0}
   527→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   528→          >
   529→            {uploading ? (
   530→              <>
   531→                <Loader2 className="h-4 w-4 animate-spin mr-2" />
   532→                Subiendo...
   533→              </>
   534→            ) : (
   535→              <>
   536→                <Upload className="h-4 w-4 mr-2" />
   537→                Subir {files.length > 0 ? `(${files.length})` : ''}
   538→              </>
   539→            )}
   540→          </Button>
   541→        </div>
   542→      </DialogContent>
   543→    </Dialog>
   544→  );
   545→}

   547→/* ──────────── File Card ──────────── */
   548→function FileCard({
   549→  file,
   550→  onDelete,
   551→}: {
   552→  file: FileItem;
   553→  onDelete: (id: string) => void;
   554→}) {
   555→  const [copied, setCopied] = useState(false);
   556→  const [deleting, setDeleting] = useState(false);
   557→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
   558→  const { toast } = useToast();
   559→
   560→  const shareUrl = getShareUrl(file.shareId);
   561→
   562→  const copyLink = async () => {
   563→    try {
   564→      await navigator.clipboard.writeText(shareUrl);
   565→      setCopied(true);
   566→      toast({ title: 'Enlace copiado', description: 'El enlace de descarga está en tu portapapeles.' });
   567→      setTimeout(() => setCopied(false), 2000);
   568→    } catch {
   569→      toast({ title: 'Error', description: 'No se pudo copiar el enlace.', variant: 'destructive' });
   570→    }
   571→  };
   572→
   573→  const handleDelete = async () => {
   574→    setDeleting(true);
   575→    try {
   576→      const res = await fetch(`/api/files?id=${file.id}`, { method: 'DELETE' });
   577→      if (res.ok) {
   578→        toast({ title: 'Eliminado', description: 'El archivo ha sido eliminado.' });
   579→        onDelete(file.id);
   580→      } else {
   581→        toast({ title: 'Error', description: 'No se pudo eliminar el archivo.', variant: 'destructive' });
   582→      }
   583→    } catch {
   584→      toast({ title: 'Error', description: 'Error de conexión.', variant: 'destructive' });
   585→    } finally {
   586→      setDeleting(false);
   587→      setShowDeleteDialog(false);
   588→    }
   589→  };
   590→
   591→  return (
   592→    <>
   593→      <motion.div

   682→/* ──────────── Dashboard Component ──────────── */
   683→function Dashboard({ user: initialUser, onLogout }: { user: User; onLogout: () => void }) {
   684→  const [user, setUser] = useState<User>(initialUser);
   685→  const [files, setFiles] = useState<FileItem[]>([]);
   686→  const [loading, setLoading] = useState(true);
   687→  const [searchTerm, setSearchTerm] = useState('');
   688→  const [stats, setStats] = useState({ totalFiles: 0, totalSize: 0, totalDownloads: 0 });
   689→  const { toast } = useToast();
   690→
   691→  const loadFiles = useCallback(async () => {
   692→    try {
   693→      const res = await fetch('/api/files');
   694→      if (res.ok) {
   695→        const data = await res.json();
   696→        setFiles(data.files);
   697→        setStats({
   698→          totalFiles: data.files.length,
   699→          totalSize: data.files.reduce((acc: number, f: FileItem) => acc + f.size, 0),
   700→          totalDownloads: data.files.reduce((acc: number, f: FileItem) => acc + f.downloads, 0),
   701→        });
   702→      }
   703→    } catch {
   704→      toast({ title: 'Error', description: 'No se pudieron cargar los archivos.', variant: 'destructive' });
   705→    } finally {
   706→      setLoading(false);
   707→    }
   708→  }, [toast]);
   709→
   710→  useEffect(() => {
   711→    loadFiles();
   712→  }, [loadFiles]);
   713→
   714→  const handleLogout = async () => {
   715→    await fetch('/api/auth/me', { method: 'POST' });
   716→    onLogout();
   717→  };
   718→
   719→  const handleDeleteFile = (id: string) => {
   720→    setFiles((prev) => prev.filter((f) => f.id !== id));
   721→    loadFiles();
   722→  };
   723→
   724→  const filteredFiles = files.filter((f) =>
   725→    f.originalName.toLowerCase().includes(searchTerm.toLowerCase())
   726→  );
   727→
   728→  return (
   729→    <div className="min-h-screen flex flex-col bg-gradient-to-br from-slate-50 to-emerald-50/30 dark:from-gray-950 dark:to-gray-900">

   734→            <div className="flex items-center justify-center w-9 h-9 bg-emerald-600 rounded-xl">
   735→              <Cloud className="h-5 w-5 text-white" />
   736→            </div>
   737→            <h1 className="text-lg font-bold tracking-tight">FileVault</h1>
   738→          </div>
   739→
   740→          <div className="flex items-center gap-3">
   741→            <div className="hidden sm:flex items-center gap-2 px-3 py-1.5 bg-muted rounded-full text-sm">
   742→              <div className="w-6 h-6 bg-emerald-600 rounded-full flex items-center justify-center text-white text-xs font-bold">
   743→                {user.username[0].toUpperCase()}
   744→              </div>
   745→              <span className="font-medium">{user.username}</span>
   746→            </div>
   747→            <Button variant="ghost" size="icon" onClick={handleLogout} className="text-muted-foreground hover:text-destructive">
   748→              <LogOut className="h-4 w-4" />
   749→            </Button>
   750→          </div>
   751→        </div>
   752→      </header>

   754→      {/* Main */}

   761→            { label: 'Descargas', value: stats.totalDownloads, icon: Download, color: 'text-violet-600' },
