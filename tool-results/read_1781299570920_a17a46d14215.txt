    77→function formatBytes(bytes: number): string {
    78→  if (bytes === 0) return '0 B';
    79→  const k = 1024;
    80→  const sizes = ['B', 'KB', 'MB', 'GB'];
    81→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    82→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
    83→}

    85→function formatDate(dateStr: string): string {
    86→  return new Date(dateStr).toLocaleDateString('es-ES', {
    87→    day: 'numeric',
    88→    month: 'short',
    89→    year: 'numeric',
    90→    hour: '2-digit',
    91→    minute: '2-digit',
    92→  });
    93→}

    95→function getFileIcon(mimeType: string) {
    96→  if (mimeType.startsWith('image/')) return <FileImage className="h-8 w-8 text-emerald-500" />;
    97→  if (mimeType.startsWith('video/')) return <FileVideo className="h-8 w-8 text-violet-500" />;
    98→  if (mimeType.startsWith('audio/')) return <FileAudio className="h-8 w-8 text-amber-500" />;
    99→  if (mimeType.includes('zip') || mimeType.includes('rar') || mimeType.includes('tar') || mimeType.includes('gz'))
   100→    return <Archive className="h-8 w-8 text-orange-500" />;
   101→  if (mimeType.includes('pdf')) return <FileText className="h-8 w-8 text-red-500" />;
   102→  if (mimeType.includes('text') || mimeType.includes('json') || mimeType.includes('xml'))
   103→    return <FileText className="h-8 w-8 text-sky-500" />;
   104→  return <File className="h-8 w-8 text-muted-foreground" />;
   105→}
   106→
   107→function getFileExtension(name: string): string {
   108→  const parts = name.split('.');
   109→  return parts.length > 1 ? parts.pop()!.toUpperCase() : 'N/A';
   110→}

   329→/* ──────────── Upload Dialog ──────────── */
   330→function UploadDialog({ onUploaded }: { onUploaded: () => void }) {
   331→  const [files, setFiles] = useState<File[]>([]);
   332→  const [uploading, setUploading] = useState(false);
   333→  const [progress, setProgress] = useState(0);
   334→  const [dragActive, setDragActive] = useState(false);
   335→  const fileInputRef = useRef<HTMLInputElement>(null);
   336→  const { toast } = useToast();
   337→
   338→  const handleFiles = (fileList: FileList | null) => {
   339→    if (!fileList) return;
   340→    setFiles((prev) => [...prev, ...Array.from(fileList)]);
   341→  };
   342→
   343→  const handleDrop = useCallback((e: React.DragEvent) => {
   344→    e.preventDefault();
   345→    setDragActive(false);
   346→    handleFiles(e.dataTransfer.files);
   347→  }, []);
   348→
   349→  const handleDragOver = (e: React.DragEvent) => {
   350→    e.preventDefault();
   351→    setDragActive(true);
   352→  };
   353→
   354→  const handleDragLeave = () => setDragActive(false);
   355→
   356→  const removeFile = (index: number) => {
   357→    setFiles((prev) => prev.filter((_, i) => i !== index));
   358→  };
   359→
   360→  const upload = async () => {
   361→    if (files.length === 0) return;
   362→    setUploading(true);
   363→    setProgress(0);
   364→
   365→    let success = 0;
   366→    let failed = 0;
   367→
   368→    for (let i = 0; i < files.length; i++) {
   369→      const formData = new FormData();
   370→      formData.append('file', files[i]);
   371→
   372→      try {
   373→        const res = await fetch('/api/files/upload', { method: 'POST', body: formData });
   374→        if (res.ok) success++;
   375→        else failed++;
   376→      } catch {
   377→        failed++;
   378→      }
   379→      setProgress(Math.round(((i + 1) / files.length) * 100));
   380→    }
   381→
   382→    setUploading(false);
   383→    setFiles([]);
   384→
   385→    if (success > 0) {
   386→      toast({
   387→        title: `${success} archivo${success > 1 ? 's' : ''} subido${success > 1 ? 's' : ''}`,
   388→        description: failed > 0 ? `${failed} archivo${failed > 1 ? 's' : ''} falló.` : 'Todos los archivos se subieron correctamente.',
   389→        variant: failed > 0 ? 'default' : 'default',
   390→      });
   391→      onUploaded();
   392→    }
   393→
   394→    if (failed > 0 && success === 0) {
   395→      toast({
   396→        title: 'Error',
   397→        description: 'No se pudieron subir los archivos.',
   398→        variant: 'destructive',
   399→      });
   400→    }
   401→  };
   402→
   403→  return (
   404→    <Dialog>
   405→      <DialogTrigger asChild>
   406→        <Button className="bg-emerald-600 hover:bg-emerald-700 text-white font-medium h-11 gap-2">
   407→          <Upload className="h-4 w-4" />
   408→          Subir archivos
   409→        </Button>
   410→      </DialogTrigger>
   411→      <DialogContent className="sm:max-w-lg">
   412→        <DialogHeader>
   413→          <DialogTitle>Subir archivos</DialogTitle>
   414→          <DialogDescription>
   415→            Arrastra archivos aquí o haz clic para seleccionar. Máximo 100MB por archivo.
   416→          </DialogDescription>
   417→        </DialogHeader>
   418→
   419→        <div
   420→          className={`border-2 border-dashed rounded-xl p-8 text-center transition-colors cursor-pointer ${
   421→            dragActive
   422→              ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30'
   423→              : 'border-muted-foreground/25 hover:border-emerald-400 hover:bg-muted/50'
   424→          }`}
   425→          onDrop={handleDrop}
   426→          onDragOver={handleDragOver}
   427→          onDragLeave={handleDragLeave}
   428→          onClick={() => fileInputRef.current?.click()}
   429→        >
   430→          <CloudUpload className="h-10 w-10 mx-auto text-muted-foreground mb-3" />
   431→          <p className="text-sm font-medium">Arrastra archivos aquí</p>
   432→          <p className="text-xs text-muted-foreground mt-1">o haz clic para seleccionar</p>
   433→          <input
   434→            ref={fileInputRef}
   435→            type="file"
   436→            multiple
   437→            className="hidden"
   438→            onChange={(e) => handleFiles(e.target.files)}
   439→          />
   440→        </div>
   441→
   442→        {files.length > 0 && (
   443→          <div className="space-y-2 max-h-48 overflow-y-auto">
   444→            {files.map((f, i) => (
   445→              <div key={`${f.name}-${i}`} className="flex items-center gap-3 p-2 rounded-lg bg-muted/50">
   446→                {getFileIcon(f.type)}
   447→                <div className="flex-1 min-w-0">
   448→                  <p className="text-sm font-medium truncate">{f.name}</p>
   449→                  <p className="text-xs text-muted-foreground">{formatBytes(f.size)}</p>
   450→                </div>
   451→                <Button
   452→                  variant="ghost"
   453→                  size="icon"
   454→                  className="h-8 w-8 shrink-0"
   455→                  onClick={(e) => {
   456→                    e.stopPropagation();
   457→                    removeFile(i);
   458→                  }}
   459→                  disabled={uploading}
   460→                >
   461→                  <X className="h-4 w-4" />
   462→                </Button>
   463→              </div>
   464→            ))}
   465→          </div>
   466→        )}
   467→
   468→        {uploading && (
   469→          <div className="space-y-2">
   470→            <Progress value={progress} className="h-2" />
   471→            <p className="text-xs text-center text-muted-foreground">{progress}% completado</p>
   472→          </div>
   473→        )}
   474→
   475→        <div className="flex justify-end gap-2">
   476→          <Button
   477→            variant="outline"
   478→            onClick={() => setFiles([])}
   479→            disabled={uploading || files.length === 0}
   480→          >
   481→            Limpiar
   482→          </Button>
   483→          <Button
   484→            onClick={upload}
   485→            disabled={uploading || files.length === 0}
   486→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   487→          >
   488→            {uploading ? (
   489→              <>
   490→                <Loader2 className="h-4 w-4 animate-spin mr-2" />
   491→                Subiendo...
   492→              </>
   493→            ) : (
   494→              <>
   495→                <Upload className="h-4 w-4 mr-2" />
   496→                Subir {files.length > 0 ? `(${files.length})` : ''}
   497→              </>
   498→            )}
   499→          </Button>
   500→        </div>
   501→      </DialogContent>
   502→    </Dialog>
   503→  );
   504→}

   506→/* ──────────── File Card ──────────── */
   507→function FileCard({
   508→  file,
   509→  onDelete,
   510→}: {
   511→  file: FileItem;
   512→  onDelete: (id: string) => void;
   513→}) {
   514→  const [copied, setCopied] = useState(false);
   515→  const [deleting, setDeleting] = useState(false);
   516→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
   517→  const { toast } = useToast();
   518→
   519→  const shareUrl = getShareUrl(file.shareId);
   520→
   521→  const copyLink = async () => {
   522→    try {
   523→      await navigator.clipboard.writeText(shareUrl);
   524→      setCopied(true);
   525→      toast({ title: 'Enlace copiado', description: 'El enlace de descarga está en tu portapapeles.' });
   526→      setTimeout(() => setCopied(false), 2000);
   527→    } catch {
   528→      toast({ title: 'Error', description: 'No se pudo copiar el enlace.', variant: 'destructive' });
   529→    }
   530→  };
   531→
   532→  const handleDelete = async () => {
   533→    setDeleting(true);
   534→    try {
   535→      const res = await fetch(`/api/files?id=${file.id}`, { method: 'DELETE' });
   536→      if (res.ok) {
   537→        toast({ title: 'Eliminado', description: 'El archivo ha sido eliminado.' });
   538→        onDelete(file.id);
   539→      } else {
   540→        toast({ title: 'Error', description: 'No se pudo eliminar el archivo.', variant: 'destructive' });
   541→      }
   542→    } catch {
   543→      toast({ title: 'Error', description: 'Error de conexión.', variant: 'destructive' });
   544→    } finally {
   545→      setDeleting(false);
   546→      setShowDeleteDialog(false);
   547→    }
   548→  };
   549→
   550→  return (
   551→    <>
   552→      <motion.div
   553→        initial={{ opacity: 0, y: 10 }}
   554→        animate={{ opacity: 1, y: 0 }}
   555→        exit={{ opacity: 0, y: -10 }}
   556→        layout
   557→        className="group"
   558→      >
   559→        <Card className="hover:shadow-md transition-shadow border-border/60">
   560→          <CardContent className="p-4">
   561→            <div className="flex items-start gap-3">
   562→              <div className="shrink-0 mt-0.5">{getFileIcon(file.mimeType)}</div>
   563→              <div className="flex-1 min-w-0">
   564→                <p className="font-medium text-sm truncate" title={file.originalName}>
   565→                  {file.originalName}
   566→                </p>
   567→                <div className="flex items-center gap-2 mt-1 flex-wrap">
   568→                  <Badge variant="secondary" className="text-xs font-normal">
   569→                    {getFileExtension(file.originalName)}
   570→                  </Badge>
   571→                  <span className="text-xs text-muted-foreground">{formatBytes(file.size)}</span>
   572→                  <span className="text-xs text-muted-foreground">•</span>
   573→                  <span className="text-xs text-muted-foreground">{formatDate(file.createdAt)}</span>
   574→                </div>
   575→                <div className="flex items-center gap-1.5 mt-1.5 text-xs text-muted-foreground">
   576→                  <Eye className="h-3 w-3" />
   577→                  <span>{file.downloads} descarga{file.downloads !== 1 ? 's' : ''}</span>
   578→                </div>
   579→              </div>
   580→            </div>

   642→function Dashboard({ user: initialUser, onLogout }: { user: User; onLogout: () => void }) {
   643→  const [user, setUser] = useState<User>(initialUser);
   644→  const [files, setFiles] = useState<FileItem[]>([]);
   645→  const [loading, setLoading] = useState(true);
   646→  const [searchTerm, setSearchTerm] = useState('');
   647→  const [stats, setStats] = useState({ totalFiles: 0, totalSize: 0, totalDownloads: 0 });
   648→  const { toast } = useToast();
   649→
   650→  const loadFiles = useCallback(async () => {
   651→    try {
   652→      const res = await fetch('/api/files');
   653→      if (res.ok) {
   654→        const data = await res.json();
   655→        setFiles(data.files);
   656→        setStats({
   657→          totalFiles: data.files.length,
   658→          totalSize: data.files.reduce((acc: number, f: FileItem) => acc + f.size, 0),
   659→          totalDownloads: data.files.reduce((acc: number, f: FileItem) => acc + f.downloads, 0),
   660→        });
   661→      }
   662→    } catch {
   663→      toast({ title: 'Error', description: 'No se pudieron cargar los archivos.', variant: 'destructive' });
   664→    } finally {
   665→      setLoading(false);
   666→    }
   667→  }, [toast]);
   668→
   669→  useEffect(() => {
   670→    loadFiles();
   671→  }, [loadFiles]);
   672→
   673→  const handleLogout = async () => {
   674→    await fetch('/api/auth/me', { method: 'POST' });
   675→    onLogout();
   676→  };
   677→
   678→  const handleDeleteFile = (id: string) => {
   679→    setFiles((prev) => prev.filter((f) => f.id !== id));
   680→    loadFiles();
   681→  };
   682→
   683→  const filteredFiles = files.filter((f) =>
   684→    f.originalName.toLowerCase().includes(searchTerm.toLowerCase())
   685→  );

   693→            <div className="flex items-center justify-center w-9 h-9 bg-emerald-600 rounded-xl">
   694→              <Cloud className="h-5 w-5 text-white" />
   695→            </div>
   696→            <h1 className="text-lg font-bold tracking-tight">FileVault</h1>
   697→          </div>
   698→
   699→          <div className="flex items-center gap-3">
   700→            <div className="hidden sm:flex items-center gap-2 px-3 py-1.5 bg-muted rounded-full text-sm">
   701→              <div className="w-6 h-6 bg-emerald-600 rounded-full flex items-center justify-center text-white text-xs font-bold">
   702→                {user.username[0].toUpperCase()}
   703→              </div>
   704→              <span className="font-medium">{user.username}</span>
   705→            </div>
   706→            <Button variant="ghost" size="icon" onClick={handleLogout} className="text-muted-foreground hover:text-destructive">
   707→              <LogOut className="h-4 w-4" />
   708→            </Button>
   709→          </div>
   710→        </div>
   711→      </header>
   712→
   713→      {/* Main */}
   714→      <main className="flex-1 max-w-6xl mx-auto w-full px-4 sm:px-6 py-6 sm:py-8">
   715→        {/* Stats */}
