    77→    77→function formatBytes(bytes: number): string {
    78→    78→  if (bytes === 0) return '0 B';
    79→    79→  const k = 1024;
    80→    80→  const sizes = ['B', 'KB', 'MB', 'GB'];
    81→    81→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    82→    82→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
    83→    83→}

   101→   101→  if (mimeType.includes('pdf')) return <FileText className="h-8 w-8 text-red-500" />;

   103→   103→    return <FileText className="h-8 w-8 text-sky-500" />;
   104→   104→  return <File className="h-8 w-8 text-muted-foreground" />;

   329→   329→/* ──────────── Upload Dialog ──────────── */
   330→   330→function UploadDialog({ onUploaded }: { onUploaded: () => void }) {
   331→   331→  const [files, setFiles] = useState<File[]>([]);
   332→   332→  const [uploading, setUploading] = useState(false);
   333→   333→  const [progress, setProgress] = useState(0);
   334→   334→  const [dragActive, setDragActive] = useState(false);
   335→   335→  const fileInputRef = useRef<HTMLInputElement>(null);
   336→   336→  const { toast } = useToast();
   337→   337→
   338→   338→  const handleFiles = (fileList: FileList | null) => {
   339→   339→    if (!fileList) return;
   340→   340→    setFiles((prev) => [...prev, ...Array.from(fileList)]);
   341→   341→  };
   342→   342→
   343→   343→  const handleDrop = useCallback((e: React.DragEvent) => {
   344→   344→    e.preventDefault();
   345→   345→    setDragActive(false);
   346→   346→    handleFiles(e.dataTransfer.files);
   347→   347→  }, []);
   348→   348→
   349→   349→  const handleDragOver = (e: React.DragEvent) => {
   350→   350→    e.preventDefault();
   351→   351→    setDragActive(true);
   352→   352→  };
   353→   353→
   354→   354→  const handleDragLeave = () => setDragActive(false);
   355→   355→
   356→   356→  const removeFile = (index: number) => {
   357→   357→    setFiles((prev) => prev.filter((_, i) => i !== index));
   358→   358→  };
   359→   359→
   360→   360→  const upload = async () => {
   361→   361→    if (files.length === 0) return;
   362→   362→    setUploading(true);
   363→   363→    setProgress(0);
   364→   364→
   365→   365→    let success = 0;
   366→   366→    let failed = 0;
   367→   367→
   368→   368→    for (let i = 0; i < files.length; i++) {
   369→   369→      const file = files[i];
   370→   370→      try {
   371→   371→        // Step 1: Get presigned URL from our API (small request, no file data)
   372→   372→        const presignRes = await fetch('/api/files/presign', {
   373→   373→          method: 'POST',
   374→   374→          headers: { 'Content-Type': 'application/json' },
   375→   375→          body: JSON.stringify({
   376→   376→            fileName: file.name,
   377→   377→            fileSize: file.size,
   378→   378→            mimeType: file.type || 'application/octet-stream',
   379→   379→          }),
   380→   380→        });
   381→   381→
   382→   382→        if (!presignRes.ok) {
   383→   383→          failed++;
   384→   384→          setProgress(Math.round(((i + 1) / files.length) * 100));
   385→   385→          continue;
   386→   386→        }
   387→   387→
   388→   388→        const { uploadUrl, r2Key } = await presignRes.json();
   389→   389→
   390→   390→        // Step 2: Upload file DIRECTLY to R2 (bypasses Vercel 4.5MB limit)
   391→   391→        const uploadRes = await fetch(uploadUrl, {
   392→   392→          method: 'PUT',
   393→   393→          body: file,
   394→   394→          headers: { 'Content-Type': file.type || 'application/octet-stream' },
   395→   395→        });
   396→   396→
   397→   397→        if (!uploadRes.ok) {
   398→   398→          failed++;
   399→   399→          setProgress(Math.round(((i + 1) / files.length) * 100));
   400→   400→          continue;
   401→   401→        }
   402→   402→
   403→   403→        // Step 3: Confirm - save metadata to database
   404→   404→        const confirmRes = await fetch('/api/files/confirm', {
   405→   405→          method: 'POST',
   406→   406→          headers: { 'Content-Type': 'application/json' },
   407→   407→          body: JSON.stringify({
   408→   408→            r2Key,
   409→   409→            originalName: file.name,
   410→   410→            mimeType: file.type || 'application/octet-stream',
   411→   411→            size: file.size,
   412→   412→          }),
   413→   413→        });
   414→   414→
   415→   415→        if (confirmRes.ok) success++;
   416→   416→        else failed++;
   417→   417→      } catch {
   418→   418→        failed++;
   419→   419→      }
   420→   420→      setProgress(Math.round(((i + 1) / files.length) * 100));
   421→   421→    }
   422→   422→
   423→   423→    setUploading(false);
   424→   424→    setFiles([]);
   425→   425→
   426→   426→    if (success > 0) {
   427→   427→      toast({
   428→   428→        title: `${success} archivo${success > 1 ? 's' : ''} subido${success > 1 ? 's' : ''}`,
   429→   429→        description: failed > 0 ? `${failed} archivo${failed > 1 ? 's' : ''} falló.` : 'Todos los archivos se subieron correctamente.',
   430→   430→        variant: failed > 0 ? 'default' : 'default',
   431→   431→      });
   432→   432→      onUploaded();
   433→   433→    }
   434→   434→
   435→   435→    if (failed > 0 && success === 0) {
   436→   436→      toast({
   437→   437→        title: 'Error',
   438→   438→        description: 'No se pudieron subir los archivos.',
   439→   439→        variant: 'destructive',
   440→   440→      });
   441→   441→    }
   442→   442→  };
   443→   443→
   444→   444→  return (
   445→   445→    <Dialog>
   446→   446→      <DialogTrigger asChild>
   447→   447→        <Button className="bg-emerald-600 hover:bg-emerald-700 text-white font-medium h-11 gap-2">
   448→   448→          <Upload className="h-4 w-4" />
   449→   449→          Subir archivos
   450→   450→        </Button>
   451→   451→      </DialogTrigger>
   452→   452→      <DialogContent className="sm:max-w-lg">
   453→   453→        <DialogHeader>
   454→   454→          <DialogTitle>Subir archivos</DialogTitle>
   455→   455→          <DialogDescription>
   456→   456→            Arrastra archivos aquí o haz clic para seleccionar. Máximo 5GB por archivo.
   457→   457→          </DialogDescription>
   458→   458→        </DialogHeader>
   459→   459→
   460→   460→        <div
   461→   461→          className={`border-2 border-dashed rounded-xl p-8 text-center transition-colors cursor-pointer ${
   462→   462→            dragActive
   463→   463→              ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30'
   464→   464→              : 'border-muted-foreground/25 hover:border-emerald-400 hover:bg-muted/50'
   465→   465→          }`}
   466→   466→          onDrop={handleDrop}
   467→   467→          onDragOver={handleDragOver}
   468→   468→          onDragLeave={handleDragLeave}
   469→   469→          onClick={() => fileInputRef.current?.click()}
   470→   470→        >
   471→   471→          <CloudUpload className="h-10 w-10 mx-auto text-muted-foreground mb-3" />
   472→   472→          <p className="text-sm font-medium">Arrastra archivos aquí</p>
   473→   473→          <p className="text-xs text-muted-foreground mt-1">o haz clic para seleccionar</p>
   474→   474→          <input
   475→   475→            ref={fileInputRef}
   476→   476→            type="file"
   477→   477→            multiple
   478→   478→            className="hidden"
   479→   479→            onChange={(e) => handleFiles(e.target.files)}
   480→   480→          />
   481→   481→        </div>
   482→   482→
   483→   483→        {files.length > 0 && (
   484→   484→          <div className="space-y-2 max-h-48 overflow-y-auto">

   548→   548→function FileCard({

   564→   564→      await navigator.clipboard.writeText(shareUrl);
